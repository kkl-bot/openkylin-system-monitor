add_library(monitor_widget_backend STATIC
    src/monitor_backend.cpp src/sampling_hub.cpp src/resource_bars.cpp)
target_include_directories(monitor_widget_backend PUBLIC src)
target_link_libraries(monitor_widget_backend PUBLIC monitor_backend Qt5::Qml Qt5::Quick)

add_library(kylinmonitorplugin MODULE src/plugin.cpp)
target_link_libraries(kylinmonitorplugin PRIVATE monitor_widget_backend)
set(WIDGET_ID org.ukui.kylin-monitor)
set(WIDGET_BUILD_DIR "${CMAKE_BINARY_DIR}/widgets/${WIDGET_ID}")
set(QML_MODULE_BUILD_DIR "${CMAKE_BINARY_DIR}/qml/org/ukui/kylinmonitor")
set_target_properties(kylinmonitorplugin PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${QML_MODULE_BUILD_DIR}")
file(MAKE_DIRECTORY "${WIDGET_BUILD_DIR}" "${QML_MODULE_BUILD_DIR}")
configure_file(qmldir "${QML_MODULE_BUILD_DIR}/qmldir" COPYONLY)
foreach(file metadata.json main.qml)
    configure_file("${file}" "${WIDGET_BUILD_DIR}/${file}" COPYONLY)
endforeach()

get_target_property(QMAKE_EXECUTABLE Qt5::qmake IMPORTED_LOCATION)
execute_process(COMMAND "${QMAKE_EXECUTABLE}" -query QT_INSTALL_QML OUTPUT_VARIABLE QT_QML_PATH OUTPUT_STRIP_TRAILING_WHITESPACE)
file(RELATIVE_PATH QML_RELATIVE_PATH "/usr" "${QT_QML_PATH}")
set(KYLIN_MONITOR_QML_INSTALL_DIR "${QML_RELATIVE_PATH}" CACHE PATH "Qt 5 QML module directory, relative to install prefix")
install(TARGETS kylinmonitorplugin LIBRARY DESTINATION "${KYLIN_MONITOR_QML_INSTALL_DIR}/org/ukui/kylinmonitor" COMPONENT Peony)
install(FILES qmldir DESTINATION "${KYLIN_MONITOR_QML_INSTALL_DIR}/org/ukui/kylinmonitor" COMPONENT Peony)
install(FILES metadata.json main.qml DESTINATION "share/ukui/widgets/${WIDGET_ID}" COMPONENT Peony)
install(FILES ../resources/kylin-monitor.svg DESTINATION "share/icons/hicolor/scalable/apps" COMPONENT Peony)
